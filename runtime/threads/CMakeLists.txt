execute_process(
  COMMAND "${PYTHON}" "${CHIP_TARGETS}" --repo "${REPO_ROOT}" --chip-pb "${CHIP_PB}" --print-thread-runtime-dirs
  OUTPUT_VARIABLE THREAD_RUNTIME_ROWS
  OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY
)
string(REPLACE "\n" ";" THREAD_RUNTIME_ROWS "${THREAD_RUNTIME_ROWS}")
set(THREAD_RUNTIME_DIR "${CMAKE_CURRENT_BINARY_DIR}" PARENT_SCOPE)
add_library(thread_runtime INTERFACE)
add_library(thread_runtime_ant INTERFACE)
foreach(PROFILE IN LISTS TARGETS)
  set(DIRECTORIES)
  foreach(ROW IN LISTS THREAD_RUNTIME_ROWS)
    string(REPLACE "|" ";" FIELDS "${ROW}")
    list(GET FIELDS 0 THREAD)
    list(GET FIELDS 1 DIRECTORY)
    if(THREAD STREQUAL PROFILE)
      list(APPEND DIRECTORIES "${DIRECTORY}")
    endif()
  endforeach()
  set(RENAME "")
  if(DIRECTORIES)
    set(THREAD_RUNTIME thread_runtime_${PROFILE})
    set(THREAD_RUNTIME_PARAMS "${ISA_DIR}/${PROFILE}")
    set(THREAD_RUNTIME_EXPORTS)
    add_library(${THREAD_RUNTIME} STATIC EXCLUDE_FROM_ALL)
    add_library(${THREAD_RUNTIME}_ant STATIC EXCLUDE_FROM_ALL)
    foreach(DIRECTORY IN LISTS DIRECTORIES)
      get_filename_component(BALL_DIRECTORY "${DIRECTORY}/../.." ABSOLUTE)
      get_filename_component(BALL "${BALL_DIRECTORY}" NAME)
      add_subdirectory("${DIRECTORY}" "${PROFILE}/${BALL}")
    endforeach()
    list(REMOVE_DUPLICATES THREAD_RUNTIME_EXPORTS)
    foreach(SYMBOL IN LISTS THREAD_RUNTIME_EXPORTS)
      string(APPEND RENAME "${SYMBOL} ${SYMBOL}__${PROFILE}\n")
      target_compile_definitions(${THREAD_RUNTIME} PRIVATE ${SYMBOL}=${SYMBOL}__${PROFILE})
      target_compile_definitions(${THREAD_RUNTIME}_ant PRIVATE ${SYMBOL}=${SYMBOL}__${PROFILE})
    endforeach()
    foreach(LIBRARY ${THREAD_RUNTIME} ${THREAD_RUNTIME}_ant)
      target_compile_options(${LIBRARY} PRIVATE -O2)
      target_compile_features(${LIBRARY} PRIVATE cxx_std_20)
      target_include_directories(${LIBRARY} PRIVATE "${THREAD_RUNTIME_PARAMS}"
                                                    "${REPO_ROOT}/bb-tests/workloads/lib"
                                                    "${BUDDY_SOURCE}/llvm/mlir/include/mlir/ExecutionEngine"
      )
    endforeach()
    target_link_libraries(${THREAD_RUNTIME} PUBLIC runtime)
    target_link_libraries(${THREAD_RUNTIME}_ant PUBLIC runtime_ant_tasks)
    target_link_libraries(thread_runtime INTERFACE ${THREAD_RUNTIME})
    target_link_libraries(thread_runtime_ant INTERFACE ${THREAD_RUNTIME}_ant)
  endif()
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${PROFILE}.rename" CONTENT "${RENAME}")
endforeach()
